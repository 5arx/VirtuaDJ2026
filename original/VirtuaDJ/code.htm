#this script simulates electricity

#and processes sound and tonearm rotation

DEF PowerBoxScript Script{

mustEvaluate TRUE

field SFNode myAudio USE myaudio

field SFNode SpeedSlider USE SpeedSlider

field SFFloat current 1

eventIn SFVec3f translation_changed

#is the arm up?

field SFBool isUp FALSE

#when lever is clicked

eventIn SFTime levertouchTime

#tonearm vertical

field SFNode ToneArmVertical USE ToneArmVertical

#tonearm horizontal

field SFNode ToneArm USE ToneArm

#is the ttable turning?

field SFBool turning FALSE

#clicking the StartStop button

eventIn SFTime SStouchtime

#static variables used to colour lights

field SFColor lightOff 0, 0, 0

field SFColor lightOn 1, 0, 0

#clicking the PowerSwitch

eventIn SFTime touchTime

#sends arm up

eventOut SFTime uptouchTime

#sends arm down

eventOut SFTime downtouchTime

#disables horiz. arm movement

#if arm is down.

eventOut SFBool sethorizsensor

#alters pitch of record

eventOut SFFloat setPitch

#clicking either speedlight

eventIn SFTime stateChange

#is the powerOn?

field SFBool powerOn FALSE

field SFBool thirtythreeenabled FALSE

#toggles timer

eventOut SFBool rotTimerenabler

# enables/disables speedlight sensors

eventOut SFBool set33Sensor

eventOut SFBool set45Sensor

#enables/disables StartStop sensor

eventOut SFBool setSStopSensor

#turns speedlights on/off

eventOut SFColor set33colour

eventOut SFColor set45colour

#turns red light output on/off

eventOut SFColor setRedpanel

#kills sound

eventOut SFTime audioToggle

url "powerbox.js"

}

]

function translation_changed (value)

{

x = SpeedSlider.translation;

p = x[1];

if (p<-32)

{

//process reductions

setPitch = current - (current*(8/100));

return;

}

if (p >32)

{ //process increase

setPitch = current + (current*(8/100));

return;

}

else if (8>p>-8)

{

setPitch = 1;

return;

}

}
